cmake_minimum_required(VERSION 3.25)
project(bgen VERSION 1.2.0 LANGUAGES C CXX)

# ─── Build options ───────────────────────────────────────────────────────────
option(BGEN_BUILD_EXAMPLES "Build example programs" OFF)
option(BGEN_BUILD_TESTS "Build tests" ON)
option(BGEN_BUILD_TOOLS "Build command-line tools (bgenix, cat-bgen, edit-bgen)" ON)
option(BGEN_WITH_S3 "Enable AWS S3 remote file support" OFF)
option(BUILD_SHARED_LIBS "Build shared libraries" OFF)

# ─── Language standards ──────────────────────────────────────────────────────
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# ─── Find dependencies ──────────────────────────────────────────────────────
find_package(ZLIB REQUIRED)
find_package(zstd REQUIRED)
find_package(SQLite3 REQUIRED)
# CMake ≥ 4.0 renamed SQLite3::SQLite3 → SQLite::SQLite3; create alias for older CMake
if(TARGET SQLite3::SQLite3 AND NOT TARGET SQLite::SQLite3)
    add_library(SQLite::SQLite3 ALIAS SQLite3::SQLite3)
elseif(TARGET SQLite::SQLite3 AND NOT TARGET SQLite3::SQLite3)
    add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()

# Use Boost's own CMake config (CMP0167 NEW — required for CMake ≥ 3.30)
cmake_policy(SET CMP0167 NEW)
find_package(Boost REQUIRED COMPONENTS
    filesystem
    thread
    date_time
    timer
    chrono
)

# ─── Optional: AWS S3 support ────────────────────────────────────────────────
if(BGEN_WITH_S3)
    find_package(AWSSDK REQUIRED COMPONENTS s3)
endif()

# ─── Generated version header ───────────────────────────────────────────────
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/bgen_revision.hpp.in"
    "${CMAKE_CURRENT_BINARY_DIR}/bgen_revision_autogenerated.hpp"
    @ONLY
)

# ─── Internal helper libraries ───────────────────────────────────────────────
add_subdirectory(db)
add_subdirectory(appcontext)

# ─── Core bgen library ───────────────────────────────────────────────────────
set(BGEN_SOURCES
    src/bgen.cpp
    src/IndexQuery.cpp
    src/MissingValue.cpp
    src/View.cpp
    src/zlib.cpp
)
if(BGEN_WITH_S3)
    list(APPEND BGEN_SOURCES src/S3StreamBuf.cpp)
endif()

add_library(bgen ${BGEN_SOURCES})
add_library(bgen::bgen ALIAS bgen)

target_include_directories(bgen
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/genfile/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE
        ${CMAKE_CURRENT_BINARY_DIR}
)
target_compile_features(bgen PUBLIC cxx_std_17)
if(BGEN_WITH_S3)
    target_compile_definitions(bgen PUBLIC BGEN_WITH_S3=1)
endif()
target_link_libraries(bgen
    PUBLIC
        ZLIB::ZLIB
        $<TARGET_NAME_IF_EXISTS:zstd::libzstd_shared>
        $<TARGET_NAME_IF_EXISTS:zstd::libzstd_static>
        SQLite3::SQLite3
        Boost::filesystem
    PRIVATE
        bgen::db
        $<$<BOOL:${BGEN_WITH_S3}>:${AWSSDK_LINK_LIBRARIES}>
)

# ─── Command-line tools ──────────────────────────────────────────────────────
if(BGEN_BUILD_TOOLS)
    add_executable(cat-bgen apps/cat-bgen.cpp)
    target_include_directories(cat-bgen PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(cat-bgen PRIVATE bgen::bgen bgen::appcontext Boost::filesystem)

    add_executable(bgenix apps/bgenix.cpp)
    target_include_directories(bgenix PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(bgenix PRIVATE bgen::bgen bgen::db bgen::appcontext Boost::filesystem)

    add_executable(edit-bgen apps/edit-bgen.cpp)
    target_include_directories(edit-bgen PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(edit-bgen PRIVATE bgen::bgen bgen::appcontext Boost::filesystem)
endif()

# ─── Examples ────────────────────────────────────────────────────────────────
if(BGEN_BUILD_EXAMPLES)
    add_executable(bgen_to_vcf example/bgen_to_vcf.cpp)
    target_link_libraries(bgen_to_vcf PRIVATE bgen::bgen)
endif()

# ─── Tests ───────────────────────────────────────────────────────────────────
if(BGEN_BUILD_TESTS)
    enable_testing()
    add_subdirectory(test)
endif()

# ─── Install & export ────────────────────────────────────────────────────────
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

# Install library targets with export set
install(TARGETS bgen bgen_db bgen_appcontext
    EXPORT bgenTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

# Install tools
if(BGEN_BUILD_TOOLS)
    install(TARGETS bgenix cat-bgen edit-bgen
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    )
endif()

# Install headers
install(DIRECTORY genfile/include/genfile
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)
install(DIRECTORY db/include/db
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)
install(DIRECTORY appcontext/include/appcontext
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

# Export targets for find_package() support
install(EXPORT bgenTargets
    FILE bgenTargets.cmake
    NAMESPACE bgen::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bgen
)

# Generate and install package config files
configure_package_config_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/bgenConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/bgenConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bgen
)
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/bgenConfigVersion.cmake"
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/bgenConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/bgenConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bgen
)
