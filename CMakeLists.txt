cmake_minimum_required(VERSION 3.25)
project(bgen VERSION 1.1.7 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Find upstream dependencies
find_package(ZLIB REQUIRED)
find_package(zstd REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Boost REQUIRED COMPONENTS
    system
    filesystem
    thread
    date_time
    timer
    chrono
)

# Generate version header
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/bgen_revision.hpp.in"
    "${CMAKE_CURRENT_BINARY_DIR}/bgen_revision_autogenerated.hpp"
    @ONLY
)

# Internal libraries
add_subdirectory(db)
add_subdirectory(appcontext)

# Core bgen library
add_library(bgen STATIC
    src/bgen.cpp
    src/IndexQuery.cpp
    src/MissingValue.cpp
    src/View.cpp
    src/zlib.cpp
)
target_include_directories(bgen
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/genfile/include>
        $<INSTALL_INTERFACE:include>
    PRIVATE
        ${CMAKE_CURRENT_BINARY_DIR}
)
target_link_libraries(bgen
    PUBLIC
        ZLIB::ZLIB
        zstd::libzstd_shared
        SQLite3::SQLite3
        Boost::filesystem
        Boost::system
    PRIVATE
        bgen_db
)

# Executables
add_executable(cat-bgen apps/cat-bgen.cpp)
target_include_directories(cat-bgen PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(cat-bgen PRIVATE bgen bgen_appcontext Boost::filesystem)

add_executable(bgenix apps/bgenix.cpp)
target_include_directories(bgenix PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(bgenix PRIVATE bgen bgen_db bgen_appcontext Boost::filesystem)

add_executable(edit-bgen apps/edit-bgen.cpp)
target_include_directories(edit-bgen PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(edit-bgen PRIVATE bgen bgen_appcontext Boost::filesystem)

# Examples (optional)
option(BGEN_BUILD_EXAMPLES "Build example programs" OFF)
if(BGEN_BUILD_EXAMPLES)
    add_executable(bgen_to_vcf example/bgen_to_vcf.cpp)
    target_link_libraries(bgen_to_vcf PRIVATE bgen)
endif()

# Tests (optional)
option(BGEN_BUILD_TESTS "Build tests" ON)
if(BGEN_BUILD_TESTS)
    enable_testing()
    add_subdirectory(test)
endif()

# Install targets
include(GNUInstallDirs)

install(TARGETS bgen bgen_db bgen_appcontext
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
)
install(TARGETS bgenix cat-bgen edit-bgen
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
install(DIRECTORY genfile/include/genfile
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)
install(DIRECTORY db/include/db
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)
install(DIRECTORY appcontext/include/appcontext
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)
